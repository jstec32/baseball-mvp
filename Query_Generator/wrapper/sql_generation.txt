Q: WHat is the Seattle Mariner's win percentage when leading after 8 innings?
A: SELECT
  COUNT(*) AS games_with_lead_after_8,
  SUM(CASE WHEN t.team_total_runs > t.opp_total_runs THEN 1 ELSE 0 END) AS held_lead_and_won,
  ROUND(
    SUM(CASE WHEN t.team_total_runs > t.opp_total_runs THEN 1 ELSE 0 END) * 1.0 /
    COUNT(*), 4
  ) AS win_pct_when_leading_after_8
FROM (
  SELECT
    g1.game_pk,
    g1.team_name,
    g1.runs_inning_1 + g1.runs_inning_2 + g1.runs_inning_3 +
    g1.runs_inning_4 + g1.runs_inning_5 + g1.runs_inning_6 +
    g1.runs_inning_7 + g1.runs_inning_8 AS runs_through_8,
    g1.team_total_runs,
    g2.team_name AS opponent,
    g2.runs_inning_1 + g2.runs_inning_2 + g2.runs_inning_3 +
    g2.runs_inning_4 + g2.runs_inning_5 + g2.runs_inning_6 +
    g2.runs_inning_7 + g2.runs_inning_8 AS opp_runs_through_8,
    g2.team_total_runs AS opp_total_runs
  FROM team_game_stats_2025 g1
  JOIN team_game_stats_2025 g2
    ON g1.game_pk = g2.game_pk AND g1.team_name != g2.team_name
  WHERE g1.team_name = 'Seattle Mariners'
) t
WHERE t.runs_through_8 > t.opp_runs_through_8;
